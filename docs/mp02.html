1 Data Acquisition

To evaluate the environmental efficiency of public transit agencies across the United States, this analysis integrates three federal datasets:

  1. Energy Information Administration (EIA) State Electricity Profiles (EIA_SEP_REPORT): Captures state-level electrical grid carbon intensity (\(\text{lbs CO}_2\text{/MWh}\)), primary generation sources, retail electricity prices, and total generation capacity.
  2. National Transit Database (NTD) 2025 Energy Consumption (NTD_ENERGY): Records annual fuel and electrical energy consumed across transit modes for every reporting US transit agency.
  3. National Transit Database (NTD) 2025 Service Provided (NTD_SERVICE): Details annual ridership via Unlinked Passenger Trips (total_upt), total Passenger Miles (sum_passenger_miles), and service area demographics.

get_eia_sep <- function(state, abbr){
  state_formatted <- str_to_lower(state) |> str_replace_all("\\s", "")
  
  dir_name <- fs::path("data", "mp02")
  file_name <- fs::path(dir_name, state_formatted)
  
  fs::dir_create(dir_name, recurse = TRUE)
  
  if(!fs::file_exists(file_name)){
    BASE_URL <- "[https://www.eia.gov](https://www.eia.gov)"
    RESPONSE <- request(BASE_URL) |> 
      req_url_path("electricity", "state", state_formatted) |>
      req_headers(
        `User-Agent` = "Mozilla/5.0 (Macintosh; Intel Mac OS X 10.15; rv:149.0) Gecko/20100101 Firefox/149.0"
      ) |>   
      req_retry(max_tries = 4) |>
      req_perform(file_name)
    
    resp_check_status(RESPONSE)
  }
  
  TABLE <- read_html(file_name) |> 
    html_element("table") |> 
    html_table() |>
    mutate(Item = str_to_lower(Item))
  
  if("U.S. rank" %in% colnames(TABLE)){
    TABLE <- TABLE |> rename(Rank = `U.S. rank`)
  }
  
  CO2_MWh <- TABLE |> 
    filter(Item == "carbon dioxide (lbs/mwh)") |>
    pull(Value) |> 
    str_replace_all(",", "") |>
    as.numeric()
  
  PRIMARY <- TABLE |> 
    filter(Item == "primary energy source") |> 
    pull(Rank)
  
  RATE <- TABLE |>
    filter(Item == "average retail price (cents/kwh)") |>
    pull(Value) |>
    as.numeric()
  
  GENERATION_MWh <- TABLE |>
    filter(Item == "net generation (megawatthours)") |>
    pull(Value) |>
    str_replace_all(",", "") |>
    as.numeric()
  
  data.frame(
    CO2_MWh               = CO2_MWh, 
    primary_source        = PRIMARY,
    electricity_price_MWh = RATE * 10, 
    generation_MWh        = GENERATION_MWh, 
    state                 = state, 
    abbreviation          = abbr
  )
}

EIA_SEP_REPORT <- map2(state.name, state.abb, get_eia_sep) |> list_rbind()

load_ntd_energy <- function(){
  DATA_DIR <- fs::path("data", "mp02")
  fs::dir_create(DATA_DIR, recurse = TRUE)

  NTD_ENERGY_FILE <- fs::path(DATA_DIR, "2025_ntd_energy.xlsx")

  if(!fs::file_exists(NTD_ENERGY_FILE)){
    request("[https://www.transit.dot.gov/](https://www.transit.dot.gov/)") |>
      req_url_path("sites",
                   "fta.dot.gov",
                   "files",
                   "2026-09",
                   "2025 Energy Consumption_260806.xlsx") |> 
      req_headers(
        `User-Agent` = "Mozilla/5.0 (Macintosh; Intel Mac OS X 10.15; rv:149.0) Gecko/20100101 Firefox/149.0") |> 
      req_perform(NTD_ENERGY_FILE)
  }
  
  if(file.info(NTD_ENERGY_FILE)$size < 1e4){
    cat("I was unable to download the NTD Energy File. Please try again.\n")
    stop("Download failed. Please delete the corrupted file by running `unlink('data/mp02/2025_ntd_energy.xlsx')` and try again.")
  }

  to_numeric_fill_0 <- function(x){
    x <- if_else(x == "-", NA, x)
    replace_na(as.numeric(x), 0)
  }

  read_excel(NTD_ENERGY_FILE) |> 
    select(-c(`Reporter Type`, 
              `Reporting Module`, 
              `Gallons of Other Fuel`, 
              `Other Fuel Description`)) |>
    mutate(across(-c(`Agency Name`, 
                     `Mode`,
                     `TOS`), 
                  to_numeric_fill_0)) |>
    group_by(`NTD ID`, `Mode`, `Agency Name`) |>
    summarize(across(where(is.numeric), sum), 
              .groups = "keep") |>
    mutate(ENERGY = sum(c_across(c(where(is.numeric))))) |>
    filter(ENERGY > 0) |>
    select(-ENERGY) |>
    ungroup()
}

NTD_ENERGY <- load_ntd_energy()

load_ntd_service <- function(){
  DATA_DIR <- fs::path("data", "mp02")
  fs::dir_create(DATA_DIR, recurse = TRUE)

  NTD_SERVICE_FILE <- fs::path(DATA_DIR, "2025_ntd_service.csv")

  if(!fs::file_exists(NTD_SERVICE_FILE)){
    request("[https://data.transportation.gov/resource/4fir-qbim.csv](https://data.transportation.gov/resource/4fir-qbim.csv)") |>
      req_url_query(`$query` = paste(
        "SELECT `_5_digit_ntd_id` AS `NTD_ID`",
        "`max_agency` AS `AGENCY`",
        "`max_reporter_type` AS `reporter_type`",
        "`max_organization_type` AS `organization_type`",
        "`max_city` AS `city`",
        "`max_state` AS `state`",
        "`mode`",
        "`type_of_service`",
        "`max_primary_uza_code` as `uza_code`",
        "`max_primary_uza_name` as `uza_name`",
        "`max_primary_uza_area_sq_miles` as `uza_sq_miles`",
        "`max_primary_uza_population` as `uza_population`",
        "`max_service_area_sq_miles` as `service_area_sq_miles`",
        "`max_service_area_population` as `service_area_population`",
        "`questionable_record` as `flagged_likely_inaccurate`",
        "`sum_unlinked_passenger_trips_upt` as `total_upt`",
        "`sum_passenger_miles` WHERE caseless_one_of(`report_year`, '2025') limit 50000",
        sep = ",")) |>
      req_headers(
        `User-Agent` = "Mozilla/5.0 (Macintosh; Intel Mac OS X 10.15; rv:149.0) Gecko/20100101 Firefox/149.0") |>
      req_perform(NTD_SERVICE_FILE)
  }

  if(file.info(NTD_SERVICE_FILE)$size < 1e4){
    cat("I was unable to download the NTD Service File. Please try again.\n")
    stop("Download failed. Please delete the corrupted file by running `unlink('data/mp02/2025_ntd_service.csv')` and try again.")
  }

  read_csv(NTD_SERVICE_FILE, show_col_types = FALSE) |>
    mutate(flagged_likely_inaccurate = !is.na(flagged_likely_inaccurate))
}

4 Multi-Table Data Analysis: Calculating Transit Emissions & The GTA Value Index

To compute the Green Transit Alliance Value Index (GTA VI), we join NTD_SERVICE, NTD_ENERGY, and EIA_SEP_REPORT across agency, mode, and state keys. We then calculate:

  1. Direct Tailpipe Emissions (\(\text{lbs CO}_2\)): Applying standard EPA/EIA combustion emission factors to each liquid and gaseous fuel reported in NTD_ENERGY:
    • Diesel Fuel: \(22.45\text{ lbs CO}_2/\text{gal}\)
    • Gasoline: \(19.60\text{ lbs CO}_2/\text{gal}\)
    • Bio-Diesel: \(21.00\text{ lbs CO}_2/\text{gal}\)
    • Compressed Natural Gas (CNG): \(13.80\text{ lbs CO}_2/\text{gal (GGE)}\)
    • Liquified Natural Gas (LNG): \(11.20\text{ lbs CO}_2/\text{gal}\)
    • Liquified Petroleum Gas (LPG): \(12.68\text{ lbs CO}_2/\text{gal}\)
    • Kerosene / Bunker Fuel: \(21.50\text{ lbs CO}_2/\text{gal}\)
    • Ethanol: \(12.70\text{ lbs CO}_2/\text{gal}\)
  2. Upstream Electrical Grid Emissions (\(\text{lbs CO}_2\)): Converting agency electrical consumption (Kilowatt hours of Electric Propulsion + Kilowatt hours of Electric Battery) from \(\text{kWh}\) to \(\text{MWh}\) (\(\div 1,000\)) and multiplying by the agency state’s CO2_MWh emission factor from EIA_SEP_REPORT.
  3. Counterfactual Driving Emissions & GTA Value Index (GTA VI):
    • Using the EPA average passenger vehicle fuel economy (\(25.4\text{ MPG}\) at \(19.60\text{ lbs CO}_2/\text{gal} \approx 0.7717\text{ lbs CO}_2\text{ per passenger-mile}\)), we estimate the counterfactual tailpipe emissions if every transit rider had instead driven their trip.
    • Net \(\text{CO}_2\) Saved (\(\text{lbs}\)) \(= \text{Counterfactual Driving Emissions} - \text{Total Transit Emissions}\).
    • Emissions Saved per Passenger Mile (\(\text{lbs CO}_2/\text{mile}\)) \(= \frac{\text{Net CO}_2\text{ Saved}}{\text{Total Passenger Miles}}\), which serves as our normalized efficiency score across Small, Medium, and Large transit agencies.

To evaluate the environmental efficiency of public transit agencies across the United States, this analysis integrates three federal datasets:

1. **Energy Information Administration (EIA) State Electricity Profiles (`EIA_SEP_REPORT`)**: Captures state-level electrical grid carbon intensity ($\text{lbs CO}_2\text{/MWh}$), primary generation sources, retail electricity prices, and total generation capacity.

## Multi-Table Data Analysis: Calculating Transit Emissions & The GTA Value Index

To compute the **Green Transit Alliance Value Index (GTA VI)**, we join `NTD_SERVICE`, `NTD_ENERGY`, and `EIA_SEP_REPORT` across agency, mode, and state keys. We then calculate:

1. **Direct Tailpipe Emissions ($\text{lbs CO}_2$)**: Applying standard EPA/EIA combustion emission factors to each liquid and gaseous fuel reported in `NTD_ENERGY`:
   * **Diesel Fuel**: $22.45\text{ lbs CO}_2/\text{gal}$
   * **Gasoline**: $19.60\text{ lbs CO}_2/\text{gal}$
   * **Bio-Diesel**: $21.00\text{ lbs CO}_2/\text{gal}$
   * **Compressed Natural Gas (CNG)**: $13.80\text{ lbs CO}_2/\text{gal (GGE)}$
   * **Liquified Natural Gas (LNG)**: $11.20\text{ lbs CO}_2/\text{gal}$
   * **Liquified Petroleum Gas (LPG)**: $12.68\text{ lbs CO}_2/\text{gal}$
   * **Kerosene / Bunker Fuel**: $21.50\text{ lbs CO}_2/\text{gal}$
   * **Ethanol**: $12.70\text{ lbs CO}_2/\text{gal}$
2. **Upstream Electrical Grid Emissions ($\text{lbs CO}_2$)**: Converting agency electrical consumption (`Kilowatt hours of Electric Propulsion` + `Kilowatt hours of Electric Battery`) from $\text{kWh}$ to $\text{MWh}$ ($\div 1,000$) and multiplying by the agency state's `CO2_MWh` emission factor from `EIA_SEP_REPORT`.
3. **Counterfactual Driving Emissions & GTA Value Index (GTA VI)**:
   * Using the EPA average passenger vehicle fuel economy ($25.4\text{ MPG}$ at $19.60\text{ lbs CO}_2/\text{gal} \approx 0.7717\text{ lbs CO}_2\text{ per passenger-mile}$), we estimate the counterfactual tailpipe emissions if every transit rider had instead driven their trip.
   * **Net $\text{CO}_2$ Saved ($\text{lbs}$)** $= \text{Counterfactual Driving Emissions} - \text{Total Transit Emissions}$.
   * **Emissions Saved per Passenger Mile ($\text{lbs CO}_2/\text{mile}$)** $= \frac{\text{Net CO}_2\text{ Saved}}{\text{Total Passenger Miles}}$, which serves as our normalized efficiency score across **Small**, **Medium**, and **Large** transit agencies.

service_by_mode <- NTD_SERVICE |>
  filter(!flagged_likely_inaccurate) |>
  group_by(NTD_ID, AGENCY, city, state, mode) |>
  summarize(

  filter(total_upt > 0, sum_passenger_miles > 0)

# Step 2: Create a state lookup from EIA_SEP_REPORT matching both abbreviation and full state name
eia_lookup <- bind_rows(
  EIA_SEP_REPORT |> mutate(state_key = str_to_upper(str_trim(abbreviation))),
  EIA_SEP_REPORT |> mutate(state_key = str_to_upper(str_trim(state)))
) |>
  distinct(state_key, .keep_all = TRUE) |>

nat_avg_price <- mean(EIA_SEP_REPORT$electricity_price_MWh, na.rm = TRUE)

# Aggregate to the Agency level and classify into Small, Medium, and Large tiers
CAR_CO2_PER_MILE_LBS <- 19.60 / 25.4 # ~0.7717 lbs CO2 per mile for average US passenger vehicle

agency_gta_scores <- transit_emissions_mode |>
  group_by(NTD_ID, AGENCY, city, state, primary_source, CO2_MWh) |>
  summarize(
    Total_UPT = sum(total_upt, na.rm = TRUE),
    Total_Passenger_Miles = sum(sum_passenger_miles, na.rm = TRUE),
    Direct_Fuel_CO2_Lbs = sum(direct_fuel_co2_lbs, na.rm = TRUE),
    Electric_CO2_Lbs = sum(upstream_electric_co2_lbs, na.rm = TRUE),
    Total_Transit_CO2_Lbs = sum(total_co2_lbs, na.rm = TRUE),
    Total_Electric_kWh = sum(total_electric_kwh, na.rm = TRUE),
    .groups = "drop"
  ) |>
  # Require meaningful commercial transit scale (at least 100,000 annual trips)
  filter(Total_UPT >= 100000, Total_Passenger_Miles >= 100000, Total_Transit_CO2_Lbs > 0) |>
  # Categorize into Small, Medium, and Large agencies based on tertiles of Total UPT
  mutate(
    Size_Category = case_when(
      Total_UPT >= quantile(Total_UPT, 2/3, na.rm = TRUE) ~ "Large Agency",
      Total_UPT >= quantile(Total_UPT, 1/3, na.rm = TRUE) ~ "Medium Agency",
      TRUE ~ "Small Agency"
    ),
    Size_Category = factor(Size_Category, levels = c("Large Agency", "Medium Agency", "Small Agency")),
    
    # Per-unit emissions
    CO2_per_UPT_Lbs = Total_Transit_CO2_Lbs / Total_UPT,
    CO2_per_Mile_Lbs = Total_Transit_CO2_Lbs / Total_Passenger_Miles,
    
    # Counterfactual Passenger Car Emissions & Net Emissions Avoided
    Counterfactual_Car_CO2_Lbs = Total_Passenger_Miles * CAR_CO2_PER_MILE_LBS,
    Net_CO2_Saved_Tons = (Counterfactual_Car_CO2_Lbs - Total_Transit_CO2_Lbs) / 2000,
    
    # GTA Value Index (GTA VI): lbs CO2 avoided per passenger mile relative to driving
    GTA_VI_Score = CAR_CO2_PER_MILE_Lbs <- (CAR_CO2_PER_MILE_LBS - CO2_per_Mile_Lbs)
  )
